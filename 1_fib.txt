# Non-Recursive Method to Generate Fibonacci Series
def fibonacci(n):
    fib_series = []  # Stores the Fibonacci series
    a = 0            # First Fibonacci number
    b = 1            # Second Fibonacci number

    # Repeat the loop n times to generate n Fibonacci numbers
    for i in range(n):
        fib_series.append(a)  # Add the current number to the series
        a, b = b, a + b       # Update a and b to the next two numbers

    return fib_series  # Return the complete Fibonacci series


# Recursive Method to Generate Fibonacci Series
def fibonacci_recursive(n):
    # Base case: If n is zero or negative, return an empty list
    if n <= 0:
        return []

    # Base case: If n is 1, return the first Fibonacci number
    elif n == 1:
        return [0]

    # Base case: If n is 2, return the first two Fibonacci numbers
    elif n == 2:
        return [0, 1]

    else:
        # Recursively generate the series containing n-1 numbers
        fib_series = fibonacci_recursive(n - 1)

        # Calculate the next number by adding the last two numbers
        fib_series.append(fib_series[-1] + fib_series[-2])

        return fib_series  # Return the updated Fibonacci series


# Main Program
# Sample Input: 7
n = int(input("Enter total numbers to print in Fibonacci series: "))

# Display the Fibonacci series using the non-recursive method
print("Fibonacci Series (Non-Recursive):", fibonacci(n))

# Display the Fibonacci series using the recursive method
print("Fibonacci Series (Recursive):    ", fibonacci_recursive(n))


# Sample Output:

# Fibonacci Series (Non-Recursive): [0, 1, 1, 2, 3, 5, 8]
# Fibonacci Series (Recursive):     [0, 1, 1, 2, 3, 5, 8]
