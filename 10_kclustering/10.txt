# STEP 1: Import the required libraries
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans


# STEP 2: Load the sales dataset and select features
# The latin1 encoding helps read this dataset if it contains special characters.
data = pd.read_csv("sales_data_sample.csv", encoding="latin1")

# Display the first five rows of the dataset.
print("First five rows:")
print(data.head())

# Select the numerical columns used for clustering and remove rows with missing values.
X = data[["QUANTITYORDERED", "PRICEEACH", "SALES"]].dropna()

# Display the number of selected records and features.
print("Selected data shape:", X.shape)


# STEP 3: Standardize the selected features
# Scaling prevents features with larger numeric ranges from dominating clustering.
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)


# STEP 4: Use the Elbow Method to help choose the number of clusters
# WCSS (within-cluster sum of squares) measures how compact the clusters are.
wcss = []

# Fit K-Means for cluster counts from 1 through 10.
for i in range(1, 11):
    model = KMeans(
        n_clusters=i,
        random_state=42,
        n_init=10
    )

    model.fit(X_scaled)
    wcss.append(model.inertia_)

# Plot WCSS against the number of clusters.
plt.figure(figsize=(8, 5))
plt.plot(range(1, 11), wcss, marker="o")
plt.xlabel("Number of Clusters")
plt.ylabel("WCSS")
plt.title("Elbow Method")
plt.tight_layout()
plt.show()


# STEP 5: Apply K-Means clustering with 3 clusters
# The original notebook uses 3 clusters for the final model.
kmeans = KMeans(
    n_clusters=3,
    random_state=42,
    n_init=10
)

# Assign a cluster label to each standardized row.
clusters = kmeans.fit_predict(X_scaled)

# Match cluster labels to the corresponding original dataset rows.
result = data.loc[X.index].copy()
result["Cluster"] = clusters

# Display a preview of the clustered dataset.
print("Clustered data:")
print(result.head())


# STEP 6: Visualize the clusters
# This scatter plot shows SALES against QUANTITYORDERED, colored by cluster.
plt.figure(figsize=(8, 5))
sns.scatterplot(
    data=result,
    x="QUANTITYORDERED",
    y="SALES",
    hue="Cluster",
    palette="viridis"
